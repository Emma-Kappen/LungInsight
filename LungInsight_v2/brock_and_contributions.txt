1109: function renderShap() {
1110:   const target = $("b-shap");
1111:   const n = V.nodules[V.selected];
1112:   if (!n) { target.textContent = "Select a nodule."; return; }
1113:   const b = brockOf(n);
1114:   if (!brockUsable(b)) {
1115:     target.innerHTML = '<div class="alert warn">Blocked: Brock result unavailable, so there is nothing to explain.</div>';
1116:     return;
1117:   }
1118:   const c = contributions(b);
1119:   const items = BROCK_PARAMS.map((p) => {
1120:     let term = "";
1121:     try { term = BROCK_TERMS[p.key](c); } catch { term = ""; }
1122:     return {p, value: p.value(c), contribution: p.contrib(c), term};
1123:   });
1124:   const intercept = c.intercept?.contribution_logit;
1125:   const peak = Math.max(...items.map((item) => Math.abs(item.contribution ?? 0)), 1e-9);
1126:   const rows = items.map(({p, value, contribution, term}) => {
1127:     const known = fin(contribution);
1128:     const width = known ? Math.abs(contribution) / peak * 48 : 0;
1129:     const bar = known && contribution !== 0 ? `<i class="shap-bar ${contribution > 0 ? "pos" : "neg"}" style="width:${width}%"></i>` : "";
1130:     return `<div class="shap-row"><span class="shap-name" title="${esc(`${p.name}: ${p.enters}`)}">${esc(p.name)}: <b>${esc(value ?? "Missing")}</b><small>${esc(term)}</small></span><div class="shap-track">${bar}</div><span class="shap-val">${known ? sgn(contribution) : "n/a"}</span></div>`;
1131:   }).join("");
1132:   const net = items.reduce((sum, item) => sum + (fin(item.contribution) ? item.contribution : 0), 0);
1133:   const lp = fin(b.logit) ? b.logit : (fin(intercept) ? intercept + net : null);
1134:   const baseline = fin(intercept) ? 1 / (1 + Math.exp(-intercept)) : null;
1135:   target.innerHTML = `<p class="hint formula" style="margin:0 0 .3rem">P = 1 / (1 + e<sup>−LP</sup>), with LP = −6.7892 + Σ the nine terms below. Each bar is that term's push on the log-odds relative to the reference nodule
1136:       (62-year-old male, no family history or emphysema, 4 mm solid nodule, lower/middle lobe, 4 nodules, not spiculated), whose LP is the intercept alone${baseline === null ? "" : ` (risk ${esc(pct(baseline))})`}.</p>
1137:     <div class="shap-legend"><span class="lo">← lowers risk</span><span class="hi">raises risk →</span></div>${rows}
1138:     <div class="shap-sum" role="group" aria-label="Sum of terms">
1139:       <span>Intercept <b>${fin(intercept) ? sgn(intercept) : "n/a"}</b></span><span>+ terms <b>${sgn(net)}</b></span>
1140:       <span>= LP <b>${lp === null ? "n/a" : sgn(lp)}</b></span><span>→ P <b>${esc(pct(b.probability))}</b></span></div>
1141:     <p class="hint">Exact for this additive model (not a sampled approximation). Model: ${esc(b.model_name || "Brock PanCan")}, ${esc(b.formula_version || "")}.</p>`;
1142: }
