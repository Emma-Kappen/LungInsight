PATIENT_ID_PATTERN = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$")
IMAGING_DIR = Path(__file__).resolve().parents[1]
from pipeline_schema import get_required, load_schema

PIPELINE_SCHEMA = load_schema(IMAGING_DIR / "stages" / "01_hu_and_crop.py")
STAGE06_PATHS = get_required(PIPELINE_SCHEMA, "paths.stage_06", expected_type=dict)
STAGE06_SCHEMA = get_required(PIPELINE_SCHEMA, "stage_06_full_ct_gradcam", expected_type=dict)
_RUNS: dict[str, dict[str, Any]] = {}
_LOCK = threading.Lock()
_RUN_ID_PATTERN = re.compile(r"^[0-9a-f]{32}$")


def _now() -> str:
    return datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")


def _public(run: dict[str, Any]) -> dict[str, Any]:
    result = dict(run)
    result.pop("_clinical_inputs", None)
    result.pop("_baseline_definition", None)
    result.pop("_process", None)
    result.pop("log_path", None)
    result.pop("contract_path", None)
    result.pop("result_path", None)
    result.pop("_state_path", None)
    return result


def _save_run(run: dict[str, Any]) -> None:
    state_path = run.get("_state_path")
    if not isinstance(state_path, str):
        return
    path = Path(state_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    saved = {key: value for key, value in run.items() if key not in {"_process", "_state_path"}}
    temporary = path.with_suffix(".tmp")
    temporary.write_text(json.dumps(saved, indent=2, allow_nan=False), encoding="utf-8")
    temporary.replace(path)


def _set(run_id: str, **values: Any) -> None:
    with _LOCK:
        _RUNS[run_id].update(values)
        _save_run(_RUNS[run_id])


def _stage_from_line(line: str) -> tuple[str, str] | None:
    match = re.search(r"=== (Stage \d{2}[^=]+) ===", line)
    if match:
        stage = match.group(1).strip()
        number = re.search(r"Stage (\d{2})", stage).group(1)
        return f"stage{number}", f"{stage} is running."
    return None


def _record_completed_stage(run_id: str, line: str) -> None:
    match = re.search(r"\[(\d{2})/07\]\s+Stage\s+(\d{2})\s+Complete", line)
    if not match or match.group(1) != match.group(2):
        return
    stage = f"stage{match.group(1)}"
    with _LOCK:
        completed = list(_RUNS[run_id].get("completed_stages", []))
    if stage not in completed:
        completed.append(stage)
        _set(run_id, completed_stages=completed)


def _analysis_brock_inputs(clinical_inputs: dict[str, Any]) -> dict[str, Any]:
    patient: dict[str, Any] = {}
    age = clinical_inputs.get("age_at_baseline_years")
    if age is not None:
        patient["age_years"] = age
    sex = clinical_inputs.get("sex_at_baseline")
    if sex in {"female", "male"}:
        patient["sex"] = sex
    family_history = clinical_inputs.get("family_history_lung_cancer")
    if family_history in {"yes", "no"}:
        patient["family_history_lung_cancer"] = family_history == "yes"
    if clinical_inputs.get("medical_history_status") == "reported":
        conditions = clinical_inputs.get("medical_history_conditions", [])
        patient["emphysema"] = "emphysema" in conditions
    return {"patient": patient, "nodules": {}}


def _stage06_records_path(patient_root: Path) -> Path:
    return patient_root / str(STAGE06_PATHS["directory"]) / str(STAGE06_SCHEMA["record_filename"])


def _write_study_result(
    run: dict[str, Any],
    *,
    run_root: Path,
    clinical_inputs: dict[str, Any],
    status: str,
    clinical_assessment: dict[str, Any] | None = None,
    failure: str | None = None,
) -> Path:
    contract_path = run_root / CONTRACT_FILENAME
    contract = None
    if contract_path.is_file():
        contract = read_json(contract_path)
        validate_contract(contract)
    stage07_path = run_root / str(get_required(PIPELINE_SCHEMA, "paths.stage_07.report_json"))
    stage07 = read_json(stage07_path) if stage07_path.is_file() else None
    nodules = contract.get("nodules", []) if isinstance(contract, dict) else []
    eligible_ids = {row["stable_nodule_id"] for row in nodules}
    if stage07 is not None:
        report_ids = {
            row.get("stable_nodule_id")
            for row in stage07.get("eligible_nodules", [])
            if isinstance(row, dict)
        }
        if report_ids != eligible_ids:
            raise ValueError(
                "Stage 07 report and validated Imaging contract contain different eligible nodule IDs."
            )

    def relative(path: Path) -> str | None:
        try:
            return path.resolve().relative_to(run_root.resolve()).as_posix() if path.is_file() else None
        except ValueError:
            return None

    eligible_nodules = []
    for nodule in nodules:
        risk = nodule["malignancy"]["brock"]
        eligible_nodules.append({
            "stable_nodule_id": nodule["stable_nodule_id"],
            "anatomical_location": nodule.get("anatomical_location"),
            "centroid_voxel": nodule["centroid_voxel"],
            "centroid_lps_mm": nodule["centroid_lps_mm"],
            "bounding_box_voxel": nodule["bbox_voxel"],
            "measurements": nodule["measurements"],
            "detection_sources": nodule["detection_sources"],
            "classification": nodule["classification"],
            "brock_risk": risk,
            "quality_status": nodule["quality_status"],
            "explanation_refs": {
                "gradcam": nodule["gradcam_refs"],
                "shap": nodule["shap_refs"],
            },
            "artifact_refs": nodule["artifact_refs"],
        })
    study = contract.get("study", {}) if isinstance(contract, dict) else {}
    errors = list(study.get("errors", [])) if isinstance(study.get("errors"), list) else []
    if failure and failure not in errors:
        errors.append(failure)
    warnings = list(study.get("warnings", [])) if isinstance(study.get("warnings"), list) else []
    result_path = run_root / f"study_result_{run['run_id']}.json"
    document = {
        "schema_version": "lunginsight-study-result-v1",
        "study_status": status,
        "created_at": _now(),
        "run": {
            "run_id": run["run_id"],
            "patient_id": run["patient_id"],
            "status": status,
            "created_at": run.get("created_at"),
            "started_at": run.get("started_at"),
            "completed_at": run.get("completed_at") or _now(),
            "completed_stages": run.get("completed_stages", []),
            "failure": failure,
        },
        "clinical_inputs": clinical_inputs,
        "study": study,
        "eligible_nodules": eligible_nodules,
        "brock_predictions": [
            {
                "stable_nodule_id": nodule["stable_nodule_id"],
                "status": nodule["brock_risk"].get("status", "unavailable"),
                "probability": nodule["brock_risk"].get("probability"),
                "missing_inputs": nodule["brock_risk"].get(
                    "features_missing",
                    nodule["brock_risk"].get("missing_inputs", []),
                ),
                "model_name": nodule["brock_risk"].get("model_name"),
                "model_version": nodule["brock_risk"].get("model_version"),
            }
            for nodule in eligible_nodules
        ],
        "clinical_assessment": clinical_assessment,
        "visualization": {
            "stage07_report": relative(stage07_path),
            "accepted_nodules_csv": relative(run_root / "07_visualization" / "nodule_report.csv"),
            "excluded_nodules_csv": relative(run_root / "07_visualization" / "excluded_outside_lung.csv"),
            "html_report": f"/api/v1/analysis/runs/{run['run_id']}/report/html",
            "stage07_artifact_refs": stage07.get("artifact_refs", {}) if isinstance(stage07, dict) else {},
        },
        "provenance": study.get("provenance", {}),
        "warnings": warnings,
        "failures": errors,
        "artifact_refs": {
            "imaging_contract": relative(contract_path),
            "stage07_report": relative(stage07_path),
            "clinical_assessment": relative(run_root / f"clinical_prediction_{run['run_id']}.json"),
            "pipeline_summary": relative(run_root / "pipeline_run_summary.json"),
            "pipeline_log": relative(run_root / f"analysis_{run['run_id']}.log"),
        },
    }
    result_path.write_text(
        json.dumps(document, indent=2, ensure_ascii=False, allow_nan=False),
        encoding="utf-8",
    )
    return result_path


def _run_pipeline(
    run_id: str,
    *,
    project_root: Path,
    patient_id: str,
    input_path: Path,
    output_root: Path,
    clinical_inputs: dict[str, Any],
    requested_malignancy_method: str | None,
    resume_from_stage02: bool = False,
    baseline_definition: str = "other_or_unverified",
) -> None:
    run_root = output_root / patient_id
    log_path = run_root / f"analysis_{run_id}.log"
    command = [
        "powershell.exe",
        "-NoProfile",
        "-ExecutionPolicy",
        "Bypass",
        "-File",
        str(project_root / "run_pipeline.ps1"),
        "-PatientId",
        patient_id,
        "-InputPatientPath",
        str(input_path),
        "-ResumeFromStage02" if resume_from_stage02 else "-RegenerateStage01",
        "-NoGui",
    ]
    try:
        run_root.mkdir(parents=True, exist_ok=True)
        _set(
            run_id,
            status="stage02_running" if resume_from_stage02 else "stage01_running",
            current_stage="stage02" if resume_from_stage02 else "stage01",
            started_at=_now(),
        )
        brock_inputs_path = run_root / f"brock_inputs_{run_id}.json"
        brock_inputs_path.write_text(
            json.dumps(_analysis_brock_inputs(clinical_inputs), indent=2, allow_nan=False),
            encoding="utf-8",
        )
        command.extend(["-BrockInputs", str(brock_inputs_path)])
        with log_path.open("w", encoding="utf-8") as log:
            process = subprocess.Popen(
                command,
                cwd=str(project_root),
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                text=True,
                encoding="utf-8",
                errors="replace",
            )
            for line in process.stdout or ():
                log.write(line)
                log.flush()
                _record_completed_stage(run_id, line)
                detected = _stage_from_line(line)
                if detected:
                    stage, _ = detected
                    _set(
                        run_id,
                        status=f"{stage}_running",
                        current_stage=stage,
                        message=None,
                    )
            exit_code = process.wait()
        if exit_code != 0:
            summary_path = run_root / "pipeline_run_summary.json"
            summary = {}
            if summary_path.is_file():
                try:
                    summary = json.loads(summary_path.read_text(encoding="utf-8-sig"))
                except json.JSONDecodeError:
                    summary = {}
            failure = summary.get("failure") or "The Imaging runner failed before completion."
            blocked = "blocked" in str(failure).lower()
            failure_message = str(failure)
            contract_path = None
            records_path = _stage06_records_path(run_root)
            if records_path.is_file() and (run_root / "meta.json").is_file():
                try:
                    contract_path = create_contract_file(
                        run_root,
                        patient_id,
                        records_path,
                        processing_status="blocked" if blocked else "failed",
                        run_id=run_id,
                        errors=[failure_message],
                    )
                except (OSError, ValueError, KeyError, TypeError) as contract_error:
                    failure_message = f"{failure_message}; partial Imaging contract could not be validated: {contract_error}"
            result_path = None
            try:
                partial_run = dict(_RUNS[run_id])
                partial_run["completed_at"] = _now()
                result_path = _write_study_result(
                    partial_run,
                    run_root=run_root,
                    clinical_inputs=clinical_inputs,
                    status="blocked" if blocked else "failed",
                    failure=failure_message,
                )
            except (OSError, ValueError, KeyError, TypeError) as result_error:
                failure_message = f"{failure_message}; structured result could not be written: {result_error}"
            _set(
                run_id,
                status="blocked" if blocked else "failed",
                current_stage=None,
                message=failure_message,
                error=failure_message,
                study_summary=build_study_summary(
                    run_root,
                    processing_status="blocked" if blocked else "failed",
                    run_id=run_id,
                    errors=[failure_message],
                ),
                artifact_refs={
                    "pipeline_log": str(log_path.relative_to(project_root).as_posix()),
                    "imaging_contract": str(contract_path.relative_to(run_root).as_posix()) if contract_path else None,
                    "study_result": str(result_path.relative_to(run_root).as_posix()) if result_path else None,
                },
                result_path=str(result_path.relative_to(run_root).as_posix()) if result_path else None,
                blocking_requirements=["Resolve automated Stage 01 mask or geometry validation failures"] if blocked else [],
                completed_at=_now(),
                exit_code=exit_code,
                log_path=str(log_path.relative_to(project_root).as_posix()),
            )
            return

        _set(run_id, status="imaging_complete", current_stage=None, completed_at=_now(), exit_code=0)
        records_path = _stage06_records_path(run_root)
        if not records_path.is_file():
            raise RuntimeError(f"Imaging completed without Stage 06 records at {records_path}.")
        contract_path = create_contract_file(
            run_root,
            patient_id,
            records_path,
            processing_status="completed",
            run_id=run_id,
        )
        contract = read_json(contract_path)
        validate_contract(contract)
        _set(run_id, status="clinical_running", current_stage="clinical", contract_path=str(contract_path.relative_to(project_root).as_posix()))
        from ...Clinical.inference import load_deployed_artifacts, run_inference

        meta_schema_version = contract["study"]["provenance"]["meta_schema_version"]
        if not isinstance(meta_schema_version, str) or not meta_schema_version:
            raise RuntimeError("The validated Imaging contract is missing its metadata schema provenance.")
        response = run_inference(
            {
                "schema_version": "1.0.0",
                "patient_id": patient_id,
                "baseline_definition": baseline_definition,
                "requested_malignancy_method": requested_malignancy_method,
                "clinician_inputs": clinical_inputs,
                "imaging_evidence": {
                    "contract": contract,
                    "source_timestamp": _now(),
                    "scan_datetime": None,
                    "gate_status": "passed",
                    "imaging_pipeline_version": f"meta_schema-{meta_schema_version}",
                    "imaging_model_versions": contract["study"]["provenance"]["imaging_model_versions"],
                },
            },
            artifacts=load_deployed_artifacts(),
        )
        result_path = run_root / f"clinical_prediction_{run_id}.json"
        result_path.write_text(json.dumps(response, indent=2), encoding="utf-8")
        current_run = dict(_RUNS[run_id])
        current_run["completed_at"] = _now()
        study_result_path = _write_study_result(
            current_run,
            run_root=run_root,
            clinical_inputs=clinical_inputs,
            status="completed",
            clinical_assessment=response,
        )
        _set(
            run_id,
            status="completed",
            current_stage=None,
            clinical_prediction=response,
            study_summary=contract["study"],
            artifact_refs={
                "imaging_contract": str(contract_path.relative_to(run_root).as_posix()),
                "clinical_prediction": str(result_path.relative_to(run_root).as_posix()),
                "study_result": str(study_result_path.relative_to(run_root).as_posix()),
                "pipeline_log": str(log_path.relative_to(run_root).as_posix()),
            },
            result_path=str(study_result_path.relative_to(run_root).as_posix()),
            completed_at=_now(),
        )
    except (OSError, ValueError, KeyError, RuntimeError, TypeError, subprocess.SubprocessError) as exc:
        message = f"Analysis failed during runner, contract validation, or Clinical inference: {exc}"
        result_path = None
        if run_root.is_dir():
            try:
                partial_run = dict(_RUNS[run_id])
                partial_run["completed_at"] = _now()
                result_path = _write_study_result(
                    partial_run,
                    run_root=run_root,
                    clinical_inputs=clinical_inputs,
                    status="failed",
                    failure=message,
                )
            except (OSError, ValueError, KeyError, TypeError) as result_error:
                message = f"{message}; structured result could not be written: {result_error}"
        _set(
            run_id,
            status="failed",
            message=message,
            error=message,
            study_summary=build_study_summary(
                run_root,
                processing_status="failed",
                run_id=run_id,
                errors=[message],
            ),
            artifact_refs={
                "pipeline_log": str(log_path.relative_to(run_root).as_posix()) if log_path.exists() else None,
                "study_result": str(result_path.relative_to(run_root).as_posix()) if result_path else None,
            },
            result_path=str(result_path.relative_to(run_root).as_posix()) if result_path else None,
            completed_at=_now(),
        )


def start_analysis(
    *,
    project_root: Path,
    output_root: Path,
    patient_id: str,
    upload_job_id: str,
    clinical_inputs: dict[str, Any],
    requested_malignancy_method: str | None,
    data_provenance: str | None,
    baseline_definition: str = "other_or_unverified",
) -> dict[str, Any]:
    if baseline_definition not in {"initial_pathologic_diagnosis", "other_or_unverified"}:
        raise ValueError("Invalid baseline_definition.")
    if not PATIENT_ID_PATTERN.fullmatch(patient_id):
        raise ValueError("Invalid patient_id.")
    upload = get_dicom_job(upload_job_id, output_root=output_root)
    if upload.get("patient_id") != patient_id:
        raise ValueError("The upload job does not belong to this patient.")
    if upload.get("status") != "preprocessing_complete":
        raise ValueError("The upload job did not complete DICOM preprocessing.")
    pipeline_input = upload.get("pipeline_input")
    if not isinstance(pipeline_input, str):
        raise ValueError("The upload job has no validated pipeline input.")
    input_path = (output_root / patient_id / pipeline_input).resolve()
    relative_parts = input_path.relative_to((output_root / patient_id).resolve()).parts
    if (
        len(relative_parts) != 4
        or relative_parts[0] != "dicom_uploads"
        or relative_parts[2] != "stage01_input"
        or relative_parts[3] != patient_id
        or not input_path.is_dir()
    ):
        raise ValueError("The upload job input is not a valid server-side artifact.")
    from ...Clinical.harmonization import validate_clinician_inputs

    validated = validate_clinician_inputs(clinical_inputs)
    if validated.get("date_of_birth"):
        # The run carries the age derived from the date of birth at submission time.
        clinical_inputs = {
            **clinical_inputs,
            "age_at_baseline_years": validated["age_at_baseline_years"],
            "age_at_baseline_years_status": "reported",
        }
    state_dir = output_root / ".analysis_runs"
    with _LOCK:
        duplicate = next(
            (
                run for run in _RUNS.values()
                if (
                    run.get("upload_job_id") == upload_job_id
                    or run.get("patient_id") == patient_id
                )
                and run.get("status") not in {"failed", "completed"}
            ),
            None,
        )
        if duplicate:
            return _public(duplicate)
        run_id = uuid.uuid4().hex
        state_path = state_dir / f"{run_id}.json"
        run = {
            "run_id": run_id,
            "upload_job_id": upload_job_id,
            "patient_id": patient_id,
            "status": "analysis_queued",
            "message": "Analysis queued for the existing operator-controlled Imaging runner.",
            "current_stage": None,
            "completed_stages": [],
            "blocking_requirements": [],
            "created_at": _now(),
            "data_provenance": data_provenance or "operator_submitted",
            "_clinical_inputs": clinical_inputs,
            "_baseline_definition": baseline_definition,
            "_state_path": str(state_path),
        }
        _RUNS[run_id] = run
        _save_run(run)
    threading.Thread(
        target=_run_pipeline,
        kwargs={
            "run_id": run_id,
            "project_root": project_root,
            "patient_id": patient_id,
            "input_path": input_path,
            "output_root": output_root,
            "clinical_inputs": clinical_inputs,
            "requested_malignancy_method": requested_malignancy_method,
            "resume_from_stage02": False,
            "baseline_definition": baseline_definition,
        },
        daemon=True,
    ).start()
    return _public(run)


def resume_analysis(run_id: str, *, project_root: Path, output_root: Path) -> dict[str, Any]:
    run = _resolve_run(run_id, output_root)
    with _LOCK:
        if run.get("status") != "awaiting_imaging_review":
            raise ValueError("This analysis is not waiting for Stage 01 review.")
        patient_id = str(run["patient_id"])
        clinical_inputs = dict(run.get("_clinical_inputs", {}))
        baseline_definition = str(run.get("_baseline_definition", "other_or_unverified"))
    patient_root = (output_root / patient_id).resolve()
    _set(run_id, status="stage02_running", current_stage="stage02", message="Resuming from Stage 02 after automated Stage 01 validation.")
    threading.Thread(
        target=_run_pipeline,
        kwargs={
            "run_id": run_id,
            "project_root": project_root,
            "patient_id": patient_id,
            "input_path": patient_root,
            "output_root": output_root,
            "clinical_inputs": clinical_inputs,
            "requested_malignancy_method": "brock",
            "resume_from_stage02": True,
            "baseline_definition": baseline_definition,
        },
        daemon=True,
    ).start()
    return _public(_RUNS[run_id])


def _resolve_run(run_id: str, output_root: Path) -> dict[str, Any]:
    with _LOCK:
        run = _RUNS.get(run_id)
        if run is not None:
            return run
    if not _RUN_ID_PATTERN.fullmatch(run_id):
        raise KeyError(run_id)
    path = output_root / ".analysis_runs" / f"{run_id}.json"
    try:
        run = json.loads(path.read_text(encoding="utf-8"))
    except FileNotFoundError as exc:
        raise KeyError(run_id) from exc
    except json.JSONDecodeError as exc:
        raise RuntimeError(f"Persisted analysis run {run_id} is invalid.") from exc
    run["_state_path"] = str(path)
    if run.get("status") in {
        "analysis_queued", "stage01_running", "stage02_running", "stage03_running",
        "stage04_running", "stage05_running", "stage06_running", "clinical_running",
    }:
        run.update({
            "status": "interrupted",
            "current_stage": None,
            "message": "The service restarted while this analysis was active; completion was not confirmed.",
            "error": "Analysis process state was lost during service restart.",
            "completed_at": _now(),
        })
        run["study_summary"] = build_study_summary(
            output_root / str(run.get("patient_id", "")),
            processing_status="interrupted",
            run_id=run_id,
            errors=[run["error"]],
        )
        _save_run(run)
    with _LOCK:
        _RUNS[run_id] = run
    return run


def get_analysis(run_id: str, *, output_root: Path) -> dict[str, Any]:
    return _public(_resolve_run(run_id, output_root))


def _run_or_raise(run_id: str, output_root: Path) -> dict[str, Any]:
    return dict(_resolve_run(run_id, output_root))


def _viewer_root(run: dict[str, Any], output_root: Path) -> Path:
    patient_id = str(run["patient_id"])
    root = (output_root / patient_id).resolve()
    if root.parent != output_root.resolve():
        raise ValueError("Invalid analysis output association.")
    return root


def _viewer_arrays(root: Path) -> tuple[np.ndarray, np.ndarray | None, np.ndarray | None]:
    volume_path = root / "volume_hu_cropped.npy"
    if not volume_path.is_file():
        volume_path = root / "volume_hu.npy"
    if not volume_path.is_file():

@lru_cache(maxsize=64)
def _cam_finite_range(path: str, mtime_ns: int) -> tuple[float, float]:
    # CT-space CAMs are NaN outside the patch coverage, so the range must ignore NaN.
    cam = np.load(path, mmap_mode="r")
    low, high = np.inf, -np.inf
    for slab in cam:
        finite = np.asarray(slab, dtype=np.float32)
        finite = finite[np.isfinite(finite)]
        if finite.size:
            low = min(low, float(finite.min()))
            high = max(high, float(finite.max()))
    return (float(low), float(high)) if np.isfinite(low) else (0.0, 0.0)


# Stage 05/06 rainbow palette (blue, green, yellow, orange, red) at positions 0, .25, .5, .75, 1.
_CAM_COLOR_STOPS = np.array(
    [[0x15, 0x38, 0xD6], [0x00, 0xA6, 0x51], [0xFF, 0xFF, 0x00], [0xFF, 0x99, 0x00], [0xD7, 0x19, 0x1C]],
    dtype=np.float32,
)


# Display polarity: for these attributes the colour map is shown inverted (red = lowest relative activation).
INVERTED_CAM_ATTRIBUTES = frozenset({"spiculation", "lobulation"})


def _cam_colormap(values: np.ndarray) -> np.ndarray:
    """Map activations in [0, 1] onto the Stage 05/06 rainbow palette."""
    position = np.clip(values, 0, 1) * (len(_CAM_COLOR_STOPS) - 1)
    lower = np.minimum(position.astype(np.int64), len(_CAM_COLOR_STOPS) - 2)
    fraction = (position - lower)[..., None]
    return _CAM_COLOR_STOPS[lower] * (1 - fraction) + _CAM_COLOR_STOPS[lower + 1] * fraction


def _volume_spacing(root: Path) -> tuple[float, float, float]:
    try:
        spacing = json.loads((root / "meta.json").read_text(encoding="utf-8-sig")).get("spacing_zyx_mm")
    except (OSError, ValueError):
        spacing = None
    if (
        isinstance(spacing, list) and len(spacing) == 3
        and all(isinstance(v, (int, float)) and not isinstance(v, bool) and np.isfinite(v) and v > 0 for v in spacing)
    ):
        return (float(spacing[0]), float(spacing[1]), float(spacing[2]))
    return (1.0, 1.0, 1.0)


def _valid_box(start: Any, end: Any) -> bool:
    return (
        isinstance(start, (list, tuple)) and isinstance(end, (list, tuple)) and len(start) == 3 and len(end) == 3
        and all(isinstance(v, (int, float)) and not isinstance(v, bool) and np.isfinite(v) for v in (*start, *end))
        and all(e > s for s, e in zip(start, end))
    )


def _ellipsoid(radius_mm: float, spacing: tuple[float, float, float]) -> np.ndarray:
    reach = [int(np.ceil(radius_mm / step)) for step in spacing]
    z, y, x = np.meshgrid(*[np.arange(-k, k + 1) for k in reach], indexing="ij")
    return (z * spacing[0]) ** 2 + (y * spacing[1]) ** 2 + (x * spacing[2]) ** 2 <= radius_mm ** 2


@lru_cache(maxsize=256)
def _refine_nodule_extent(
    volume_path: str,
    lung_path: str,
    mtime_ns: int,
    start: tuple[float, ...],
    end: tuple[float, ...],
    spacing: tuple[float, float, float],
) -> dict[str, Any] | None:
    """Grow the detector box to the full extent of the nodule seen in the CT.

    The nodule is the connected component of dense voxels that overlaps the detector box. Vessels and wall
    attachments are separated by a morphological opening sized to the nodule, then edges are restored. Growth is
    bounded by a margin proportional to the detected size, and the result always contains the detector box.
    """
    from scipy import ndimage

    volume = np.load(volume_path, mmap_mode="r")
    step = np.asarray(spacing, dtype=np.float64)
    det_start = np.floor(np.asarray(start)).astype(int)
    det_end = np.ceil(np.asarray(end)).astype(int)
    det_mm = (det_end - det_start) * step
    inplane = float(max(det_mm[1], det_mm[2]))
    margin_mm = np.array([np.clip(0.5 * det_mm[0], 2.5, 7.5), np.clip(0.5 * inplane, 3.0, 8.0), np.clip(0.5 * inplane, 3.0, 8.0)])
    margin = np.ceil(margin_mm / step).astype(int)
    low = np.maximum(det_start - margin, 0)
    high = np.minimum(det_end + margin, np.asarray(volume.shape))
    window = tuple(slice(int(a), int(b)) for a, b in zip(low, high))
    region = np.asarray(volume[window], dtype=np.float32)
    seed = np.zeros(region.shape, dtype=bool)
    seed[tuple(slice(int(max(a - l, 0)), int(b - l)) for a, b, l in zip(det_start, det_end, low))] = True
    if not seed.any():
        return None
    core = float(np.percentile(region[seed], 90))
    background = float(np.percentile(region, 20))
    if core - background < 150.0:
        return None  # no tissue denser than the surrounding lung inside the detector box
    threshold = float(np.clip((core + background) / 2.0, -700.0, -300.0))
    mask = region > threshold
    if lung_path:
        lung = np.asarray(np.load(lung_path, mmap_mode="r")[window], dtype=bool)
        mask &= ndimage.binary_dilation(lung, structure=np.ones((1, 3, 3), dtype=bool), iterations=3)
    radius = max(1.2, 0.35 * inplane)
    component = None
    used_radius = radius
    for candidate in (radius, radius * 0.7, radius * 0.5, 1.0):
        structure = _ellipsoid(candidate, spacing)
        labels, count = ndimage.label(ndimage.binary_opening(mask, structure=structure), structure=np.ones((3, 3, 3)))
        if count == 0:
            continue
        overlap = np.bincount(labels[seed & (labels > 0)], minlength=count + 1)
        overlap[0] = 0
        if overlap.max() == 0:
            continue
        component = ndimage.binary_dilation(labels == overlap.argmax(), structure=structure) & mask
        used_radius = candidate
        break
    if component is None or not component.any():
        return None
    found = np.argwhere(component)
    comp_start = low + found.min(axis=0)
    comp_end = low + found.max(axis=0) + 1
    slice_boxes: dict[str, list[int]] = {}
    for z in np.flatnonzero(component.any(axis=(1, 2))):
        ys, xs = np.nonzero(component[z])
        slice_boxes[str(int(low[0] + z))] = [
            int(low[1] + ys.min()), int(low[2] + xs.min()), int(low[1] + ys.max() + 1), int(low[2] + xs.max() + 1),
        ]
    return {
        "start": [int(v) for v in np.minimum(comp_start, det_start)],
        "end": [int(v) for v in np.maximum(comp_end, det_end)],
        "slice_boxes": slice_boxes,
        "threshold_hu": round(threshold, 1),
        "opening_radius_mm": round(float(used_radius), 2),
    }


def _nodule_extent(root: Path, nodule: dict[str, Any], spacing: tuple[float, float, float]) -> dict[str, Any] | None:
    bbox = nodule.get("bbox_voxel") or {}
    start, end = bbox.get("start_zyx"), bbox.get("end_zyx_exclusive")
    if not _valid_box(start, end):
        return None
    volume_path = root / "volume_hu_cropped.npy"
    if not volume_path.is_file():
        volume_path = root / "volume_hu.npy"
    lung_path = root / "lung_mask.npy"
    start_key = tuple(float(v) for v in start)
    end_key = tuple(float(v) for v in end)
    try:
        refined = _refine_nodule_extent(
            str(volume_path), str(lung_path) if lung_path.is_file() else "",
            volume_path.stat().st_mtime_ns, start_key, end_key, spacing,
        )
    except (OSError, ValueError):
        refined = None
    if refined is None:
        return {
            "method": "detector_box",
            "start": [int(np.floor(v)) for v in start], "end": [int(np.ceil(v)) for v in end], "slice_boxes": {},
        }
    return {"method": "ct_region_growing_union_detector", **refined}


MANUAL_BROCK_FIELDS = frozenset({
    "age_years", "sex", "family_history_lung_cancer", "emphysema",
    "nodule_size_mm", "nodule_type", "upper_lobe", "nodule_count", "spiculation",
})
_NODULE_ID_PATTERN = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$")


class BrockConflictError(ValueError):
    """A supplied value would overwrite a value that already has a known source."""


def _manual_brock_path(root: Path, run_id: str) -> Path:
    return root / f"brock_manual_{run_id}.json"


def _read_manual_brock(root: Path, run_id: str) -> dict[str, Any]:
    path = _manual_brock_path(root, run_id)
    if not path.is_file():
        return {}
    try:
        stored = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError) as exc:
        raise RuntimeError(f"Stored clinician-supplied Brock inputs for run {run_id} are unreadable.") from exc
    nodules = stored.get("nodules") if isinstance(stored, dict) else None
    return nodules if isinstance(nodules, dict) else {}


def compute_manual_brock(
    run_id: str,
    nodule_id: str,
    supplied: dict[str, Any],
    *,
    output_root: Path,
) -> dict[str, Any]:
    """Compute Brock for a nodule whose Stage 04 result was withheld, using clinician-supplied missing inputs.

    Only inputs that have no known source value can be supplied; a known value is never overwritten. The result is
    stored beside the run and is clearly separate from the validated Imaging contract.
    """
    from brock import estimate_brock

    if not _NODULE_ID_PATTERN.fullmatch(nodule_id):
        raise ValueError("Invalid nodule identifier.")
    run = _run_or_raise(run_id, output_root)
    root = _viewer_root(run, output_root)
    if not supplied:
        raise ValueError("No Brock inputs were supplied.")
    unknown = sorted(set(supplied) - MANUAL_BROCK_FIELDS)
    if unknown:
        raise ValueError(f"Unsupported Brock inputs: {unknown}.")
    predictions_path = root / "04_malignancy_prediction" / "malignancy_predictions.json"
    if not predictions_path.is_file():
        raise FileNotFoundError("Stage 04 has not produced Brock inputs for this run.")
    payload = read_json(predictions_path)
    row = next(
        (item for item in payload.get("predictions", []) if item.get("stable_nodule_id") == nodule_id),
        None,
    )
    if row is None:
        raise KeyError(nodule_id)
    provenance = row.get("input_provenance") or {}
    sourced = {**(provenance.get("patient_fields") or {}), **(provenance.get("nodule_fields") or {})}
    values = {name: (sourced.get(name) or {}).get("value") for name in MANUAL_BROCK_FIELDS}
    already_known = sorted(name for name in supplied if values.get(name) is not None)
    if already_known:
        raise BrockConflictError(
            f"{already_known} already have known source values and are not overwritten; "
            "only missing inputs can be supplied."
        )
    stored = _read_manual_brock(root, run_id)
    merged = {**(stored.get(nodule_id, {}).get("supplied") or {}), **supplied}
    values.update(merged)
    result = estimate_brock(values)
    if result.get("status") == "incomplete":
        raise ValueError(f"Brock still needs: {result.get('features_missing')}.")
    if result.get("status") != "available":
        raise ValueError(result.get("failure_reason") or "The supplied Brock inputs are invalid.")
    record = {
        "supplied": merged,
        "supplied_at": _now(),
        "brock": result,
        "provenance": {
            "source": "clinician_supplied_after_run",
            "note": "Inputs without a derivable source value were entered by the user after the run; "
                    "the validated Imaging contract is unchanged.",
        },
    }
    stored[nodule_id] = record
    path = _manual_brock_path(root, run_id)
    temporary = path.with_suffix(".tmp")
    temporary.write_text(
        json.dumps({"schema_version": "brock-manual-v1", "run_id": run_id, "nodules": stored}, indent=2, allow_nan=False),
        encoding="utf-8",
    )
    temporary.replace(path)
    return record


def render_viewer_slice(
    run_id: str,
    index: int,
    *,
    output_root: Path,
    lung_overlay: bool = False,
    heart_overlay: bool = False,
    gradcam_nodule_ids: Sequence[str] | None = None,
    gradcam_nodule_id: str | None = None,
    gradcam_attribute: str | None = None,
    gradcam_alpha: float = 0.55,
) -> bytes:
    run = _run_or_raise(run_id, output_root)
    root = _viewer_root(run, output_root)
    volume, lung, heart = _viewer_arrays(root)
    if index < 0 or index >= volume.shape[0]:
        raise IndexError("Slice index is outside the current volume.")
    if not 0.0 <= gradcam_alpha <= 1.0:
        raise ValueError("gradcam_alpha must be between 0 and 1.")
    pixels = np.asarray(volume[index], dtype=np.float32)
    level, width = -600.0, 1500.0
    gray = np.clip((pixels - (level - width / 2)) / width * 255, 0, 255).astype(np.uint8)
    rgb = np.repeat(gray[:, :, None], 3, axis=2)
    if lung_overlay and lung is not None and lung.shape == volume.shape:
        mask = np.asarray(lung[index], dtype=bool)
        rgb[mask, 1] = np.maximum(rgb[mask, 1], 150)
    if heart_overlay and heart is not None and heart.shape == volume.shape:
        mask = np.asarray(heart[index], dtype=bool)
        rgb[mask, 0] = 220
        rgb[mask, 1] = np.minimum(rgb[mask, 1], 80)
    selected_ids = list(gradcam_nodule_ids or ([gradcam_nodule_id] if gradcam_nodule_id else []))
    if selected_ids and gradcam_attribute:
        spacing = _volume_spacing(root)
        contract_path = root / CONTRACT_FILENAME
        if contract_path.is_file():
            contract = read_json(contract_path)
            validate_contract(contract)
            strength: np.ndarray | None = None
            covered: np.ndarray | None = None
            for nodule in contract.get("nodules", []):
                if nodule.get("stable_nodule_id") not in selected_ids:
                    continue
                reference = (nodule.get("gradcam_refs") or {}).get(gradcam_attribute, {})
                cam_path = reference.get("ct_cam_path") if isinstance(reference, dict) else None
                if not isinstance(cam_path, str):
                    continue
                resolved = (root / cam_path).resolve()
                try:
                    resolved.relative_to(root)
                except ValueError:
                    continue
                if not resolved.is_file():
                    continue
                cam = np.load(resolved, mmap_mode="r")
                if cam.ndim != 3 or cam.shape != volume.shape:
                    continue
                activation = np.asarray(cam[index], dtype=np.float32)
                finite = np.isfinite(activation)
                if not finite.any():
                    continue
                # Same per-nodule min-max convention as the Stage 05/06 Grad-CAM images. The CAM covers the
                # nodule-centred 64^3 patch, so it is drawn as that full square at constant opacity (no fading).
                low, high = _cam_finite_range(str(resolved), resolved.stat().st_mtime_ns)
                normalized = (
                    np.clip((np.where(finite, activation, low) - low) / (high - low), 0.0, 1.0)
                    if high - low > 1e-12
                    else np.full(activation.shape, 0.5, dtype=np.float32)
                )
                normalized = np.where(finite, normalized, 0.0)
                if gradcam_attribute in INVERTED_CAM_ATTRIBUTES:
                    normalized = np.where(finite, 1.0 - normalized, 0.0)
                if strength is None:
                    strength, covered = normalized, finite
                else:
                    strength = np.where(finite & (~covered | (normalized > strength)), normalized, strength)
                    covered = covered | finite
            if strength is not None and covered is not None:
                heat = _cam_colormap(strength)
                opacity = (covered * gradcam_alpha)[..., None]
                rgb = (rgb * (1 - opacity) + heat * opacity).astype(np.uint8)
    image = Image.fromarray(rgb, mode="RGB")
    output = io.BytesIO()
    image.save(output, format="PNG", optimize=True)
    return output.getvalue()

def write_text(path, content):
    path.write_text(content, encoding="utf-8")

def clean_header(value):
    return str(value).strip().lstrip("\ufeff")

def unique_headers(headers):
    counts = {}
    result = []

    for raw in headers:
        name = clean_header(raw)
        if not name:
            name = "unnamed_column"

        key = name.lower()
        counts[key] = counts.get(key, 0) + 1

        if counts[key] == 1:
            result.append(name)
        else:
            result.append(f"{name}__duplicate_{counts[key]}")

    return result

def find_column(headers, candidates, description):
    lower_map = {h.lower(): h for h in headers}

    for candidate in candidates:
        if candidate.lower() in lower_map:
            return lower_map[candidate.lower()]

    raise RuntimeError(
        f"Required field not found for {description}. "
        f"Tried {candidates}. Available fields: {headers}"
    )

def is_metadata_row(row):
    values = [str(x).strip() for x in row if x is not None]
    joined = " ".join(values).upper()

    if any("CDE_ID:" in value.upper() for value in values):
        return True

    metadata_markers = (
        "BCR_PATIENT_BARCODE",
        "GENDER",
        "RACE",
        "TOBACCO_SMOKING",
        "CDE_ID",
    )

    if any(marker in joined for marker in metadata_markers):
        return True

    return False

def safe_numeric(value):
    if value is None:
        return np.nan

    value = str(value).strip()

    if not value:
        return np.nan

    if value.lower() in {
        "na", "n/a", "unknown", "not reported",
        "not available", "not evaluated", "[unknown]",
        "discrepancy", "null", "none", "--"
    }:
        return np.nan

    try:
        number = float(value)
        if not math.isfinite(number):
            return np.nan
        return number
    except (TypeError, ValueError):
        return np.nan

def normalize_category(value):

def parse_tcga_file(path, cohort):
    """
    Find the actual TSV header by the barcode field.
    Preserve duplicate column names positionally.
    Accept only rows with a valid TCGA patient barcode.
    Count metadata and invalid-identity rows separately.
    Never export patient identifiers.
    """

    reconciliation = {
        "cohort": cohort,
        "file": path.name,
        "header_row_index_zero_based": None,
        "raw_rows_after_header": 0,
        "valid_patient_rows": 0,
        "metadata_rows_excluded": 0,
        "invalid_or_blank_identity_rows_excluded": 0,
        "duplicate_patient_barcodes": 0,
        "row_width_mismatch": 0,
    }

    with path.open("r", encoding="utf-8-sig", newline="") as handle:
        reader = csv.reader(handle, delimiter="\t")
        all_rows = list(reader)

    header_index = None

    for index, row in enumerate(all_rows):
        normalized = [clean_header(x).lower() for x in row]
        if "bcr_patient_barcode" in normalized:
            header_index = index
            break

    if header_index is None:
        raise RuntimeError(
            f"{cohort}: could not locate the bcr_patient_barcode header row."
        )

    reconciliation["header_row_index_zero_based"] = header_index

    original_headers = [clean_header(x) for x in all_rows[header_index]]
    headers = unique_headers(original_headers)

    barcode_index = next(
        i for i, value in enumerate(original_headers)
        if value.lower() == "bcr_patient_barcode"
    )

    records = []
    seen_barcodes = set()

    for row in all_rows[header_index + 1:]:
        if not row or not any(str(x).strip() for x in row):
            continue

        reconciliation["raw_rows_after_header"] += 1

        if len(row) != len(headers):
            reconciliation["row_width_mismatch"] += 1
            continue

        barcode = str(row[barcode_index]).strip()

        if not PATIENT_BARCODE.fullmatch(barcode):
            if is_metadata_row(row):
                reconciliation["metadata_rows_excluded"] += 1
            else:
                reconciliation["invalid_or_blank_identity_rows_excluded"] += 1
            continue

        barcode_key = barcode.upper()

        if barcode_key in seen_barcodes:
            reconciliation["duplicate_patient_barcodes"] += 1
            continue

        seen_barcodes.add(barcode_key)

        # Positional construction prevents duplicate TSV headers
        # from silently overwriting one another.
        record = {}
        for i, header in enumerate(headers):
            record[header] = row[i].strip() if row[i] is not None else ""

        # Keep the barcode only in memory for deduplication.
        # It is never included in reports, model features, or exports.
        records.append(record)

    reconciliation["valid_patient_rows"] = len(records)

    if reconciliation["row_width_mismatch"] > 0:
        raise RuntimeError(
            f"{cohort}: found {reconciliation['row_width_mismatch']} rows "
            "with a field count inconsistent with the header. "
            "Parser stopped rather than silently discarding these rows."
        )

    if reconciliation["duplicate_patient_barcodes"] > 0:
        raise RuntimeError(
            f"{cohort}: duplicate patient barcodes detected. "
            "Parser stopped; no patient records were silently deduplicated."
        )

    expected = EXPECTED[cohort]
    if len(records) != expected:
        raise RuntimeError(
            f"{cohort}: corrected parser recovered {len(records)} valid "
            f"patients; expected exactly {expected}. Model training blocked."
        )

    return records, headers, reconciliation

    )

    selected_fields = {
        "LUAD": {
            "age": age_luad,
            "sex": sex_luad,
            "smoking_status": smoking_luad,
            "pack_years": packyears_luad,
            "race_sensitivity": race_luad,
        },
        "LUSC": {
            "age": age_lusc,
            "sex": sex_lusc,
            "smoking_status": smoking_lusc,
            "pack_years": packyears_lusc,
            "race_sensitivity": race_lusc,
        },
    }

    # --------------------------------------------------------
    # Harmonize only the four confirmed primary predictors.
    # Race is retained in a separate sensitivity column.
    # --------------------------------------------------------

    def make_frame(records, cohort, mapping):
        rows = []

        for record in records:
            age = safe_numeric(record.get(mapping["age"]))
            sex = normalize_category(record.get(mapping["sex"]))
            smoking = normalize_category(record.get(mapping["smoking_status"]))
            pack_years = safe_numeric(record.get(mapping["pack_years"]))
            race = normalize_category(record.get(mapping["race_sensitivity"]))

            # Smoking status codes are categorical, not ordinal.
            # Codes 1-5 are retained as category labels.
            # Unknown, undocumented, discrepancy, and other codes
            # are treated as missing rather than assigned an order.
            if not pd.isna(smoking):
                smoking_text = str(smoking).strip()
                if smoking_text.endswith(".0"):
                    smoking_text = smoking_text[:-2]

                if smoking_text in {"1", "2", "3", "4", "5"}:
                    smoking = smoking_text
                else:
                    smoking = np.nan

            rows.append({
                "cohort": cohort,
                "target": 1 if cohort == "LUAD" else 0,
                "age_at_diagnosis": age,
                "sex": sex,
                "smoking_status": smoking,
                "pack_years": pack_years,
                "race_sensitivity": race,
            })

        return pd.DataFrame(rows)

    df = pd.concat(
        [
            make_frame(luad_records, "LUAD", selected_fields["LUAD"]),
            make_frame(lusc_records, "LUSC", selected_fields["LUSC"]),
        ],
        ignore_index=True
    )

    if len(df) != EXPECTED_TOTAL:
        raise RuntimeError(
            f"Combined dataset has {len(df)} rows; expected {EXPECTED_TOTAL}."
        )

    if df["target"].nunique() != 2:
        raise RuntimeError("Both target classes are required.")

    if df["cohort"].value_counts().to_dict() != EXPECTED:
        raise RuntimeError("Combined cohort counts do not match expected counts.")

    primary_features = [
        "age_at_diagnosis",
        "sex",
        "smoking_status",
        "pack_years",
    ]
    race_features = primary_features + ["race_sensitivity"]

    # Guard against accidental forbidden fields entering predictors.
    for feature in primary_features + ["race_sensitivity"]:
        lower = feature.lower()
        if any(token in lower for token in FORBIDDEN_PREDICTOR_TOKENS):
            raise RuntimeError(f"Forbidden predictor field detected: {feature}")

    # --------------------------------------------------------
    # Corrected parser reconciliation and feature missingness
    # --------------------------------------------------------

    pd.DataFrame([luad_recon, lusc_recon]).to_csv(
        RECONCILIATION_PATH, index=False
    )

    missingness_rows = []
    for cohort, subset in df.groupby("cohort"):
        for feature in primary_features + ["race_sensitivity"]:
            missing_count = int(subset[feature].isna().sum())
            missingness_rows.append({
                "cohort": cohort,
                "feature": feature,
                "patient_count": int(len(subset)),
                "missing_count": missing_count,
                "observed_count": int(len(subset) - missing_count),
                "missing_percent": round(
                    100.0 * missing_count / len(subset), 2
                ),
            })

    pd.DataFrame(missingness_rows).to_csv(
        MISSINGNESS_PATH, index=False
    )

    # Fail if a primary feature is entirely missing in either cohort.
    for cohort, subset in df.groupby("cohort"):
        for feature in primary_features:
            if subset[feature].notna().sum() == 0:
                raise RuntimeError(
                    f"{cohort}: primary feature '{feature}' is entirely missing. "
                    "Model training blocked."
                )

    # --------------------------------------------------------
    # Build preprocessing pipelines
    # --------------------------------------------------------

    numeric_features = ["age_at_diagnosis", "pack_years"]
    categorical_features = ["sex", "smoking_status"]

    def make_one_hot_encoder():
        try:
            return OneHotEncoder(
                handle_unknown="ignore",
                sparse_output=False
            )
        except TypeError:
            return OneHotEncoder(
                handle_unknown="ignore",
                sparse=False
            )

    def make_pipeline(algorithm, use_race):
        categorical = list(categorical_features)
        if use_race:
            categorical.append("race_sensitivity")

        numeric_pipeline = Pipeline([
            ("imputer", SimpleImputer(strategy="median")),
            ("scaler", StandardScaler()),
        ])

        categorical_pipeline = Pipeline([
            ("imputer", SimpleImputer(
                strategy="constant",
                fill_value="__MISSING__"
            )),
            ("onehot", make_one_hot_encoder()),
        ])

        preprocessor = ColumnTransformer([
            ("numeric", numeric_pipeline, numeric_features),
            ("categorical", categorical_pipeline, categorical),
        ], remainder="drop")

        if algorithm == "logistic_regression":
            estimator = LogisticRegression(
                max_iter=5000,
                class_weight="balanced",
                solver="liblinear",
                random_state=RANDOM_STATE,
            )
        elif algorithm == "random_forest":
            estimator = RandomForestClassifier(
                n_estimators=500,
                min_samples_leaf=3,
                class_weight="balanced_subsample",
                random_state=RANDOM_STATE,
                n_jobs=-1,
            )
        else:
            raise ValueError(f"Unknown algorithm: {algorithm}")

        return Pipeline([
            ("preprocessor", preprocessor),
            ("classifier", estimator),
        ])

    # --------------------------------------------------------
    # Metrics
    # --------------------------------------------------------

    def calculate_metrics(y_true, probabilities, predictions):
        tn, fp, fn, tp = confusion_matrix(
            y_true, predictions, labels=[0, 1]
        ).ravel()

        sensitivity = tp / (tp + fn) if (tp + fn) else np.nan
        specificity = tn / (tn + fp) if (tn + fp) else np.nan

        return {
            "roc_auc": float(roc_auc_score(y_true, probabilities)),
            "average_precision": float(
                average_precision_score(y_true, probabilities)
            ),
            "accuracy": float(accuracy_score(y_true, predictions)),
            "balanced_accuracy": float(
                balanced_accuracy_score(y_true, predictions)
            ),
            "sensitivity_luad": float(sensitivity),
            "specificity_lusc": float(specificity),
            "true_negative_lusc": int(tn),
            "false_positive_luad": int(fp),
            "false_negative_lusc": int(fn),
            "true_positive_luad": int(tp),
        }

    # Fixed patient-level split. This is an exploratory split only;
    # cohort and target remain perfectly confounded.
    train_idx, test_idx = train_test_split(
        np.arange(len(df)),
        test_size=TEST_SIZE,
        random_state=RANDOM_STATE,
        stratify=df["target"].to_numpy(),
    )

    y = df["target"].to_numpy()
    results = []
    final_models = {}

    configurations = [
        ("primary_four_predictors", False),
        ("race_sensitivity", True),
    ]

    algorithms = [
        "logistic_regression",
        "random_forest",
    ]

    cv = StratifiedKFold(
        n_splits=CV_FOLDS,
        shuffle=True,
        random_state=RANDOM_STATE,
    )

    for feature_set_name, use_race in configurations:
        features = race_features if use_race else primary_features
        X = df[features].copy()

        for algorithm in algorithms:
            pipeline = make_pipeline(algorithm, use_race)

            X_train = X.iloc[train_idx]
            X_test = X.iloc[test_idx]
            y_train = y[train_idx]
            y_test = y[test_idx]

            pipeline.fit(X_train, y_train)

            test_probabilities = pipeline.predict_proba(X_test)[:, 1]
            test_predictions = (test_probabilities >= 0.5).astype(int)

            test_metrics = calculate_metrics(
                y_test,
                test_probabilities,
                test_predictions,
            )

            # Out-of-fold estimates are included for transparency.
            # They are not independent validation because the cohort
            # label is perfectly aligned with the target.
            cv_probabilities = cross_val_predict(
                make_pipeline(algorithm, use_race),
                X,
                y,
                cv=cv,
                method="predict_proba",
                n_jobs=1,
            )[:, 1]

            cv_predictions = (cv_probabilities >= 0.5).astype(int)

            cv_metrics = calculate_metrics(
                y,
                cv_probabilities,
                cv_predictions,
            )

            results.append({
                "feature_set": feature_set_name,
                "algorithm": algorithm,
                "validation_interpretation": "COHORT-CONFOUNDED; NOT GENERALISABLE",
                "test_patient_count": int(len(test_idx)),
                "test_roc_auc": test_metrics["roc_auc"],
                "test_average_precision": test_metrics["average_precision"],
                "test_accuracy": test_metrics["accuracy"],
                "test_balanced_accuracy": test_metrics["balanced_accuracy"],
                "test_sensitivity_luad": test_metrics["sensitivity_luad"],
                "test_specificity_lusc": test_metrics["specificity_lusc"],
                "test_true_negative_lusc": test_metrics["true_negative_lusc"],
                "test_false_positive_luad": test_metrics["false_positive_luad"],
                "test_false_negative_lusc": test_metrics["false_negative_lusc"],
                "test_true_positive_luad": test_metrics["true_positive_luad"],
                "cv_roc_auc": cv_metrics["roc_auc"],
                "cv_average_precision": cv_metrics["average_precision"],
                "cv_accuracy": cv_metrics["accuracy"],
                "cv_balanced_accuracy": cv_metrics["balanced_accuracy"],
                "cv_sensitivity_luad": cv_metrics["sensitivity_luad"],
                "cv_specificity_lusc": cv_metrics["specificity_lusc"],
            })

            # Fit the final deployable research artifact on all available
            # TCGA records. This does not connect it to the application API.
            final_pipeline = make_pipeline(algorithm, use_race)
            final_pipeline.fit(X, y)

            final_models[f"{feature_set_name}__{algorithm}"] = {
                "pipeline": final_pipeline,
                "features": features,
                "feature_set": feature_set_name,
                "algorithm": algorithm,
                "uses_race": bool(use_race),
            }

    metrics_df = pd.DataFrame(results)
    metrics_df.to_csv(METRICS_PATH, index=False)

    # --------------------------------------------------------
    # Independent model artifact
    # --------------------------------------------------------

    artifact = {
        "model_name": "LungInsight TCGA LUAD-LUSC Research Histology Model",
        "model_version": "research-v3",
        "created_utc": datetime.now(timezone.utc).isoformat(),
        "target": {
            "name": "cancer_type",
            "classes": {
                "0": "LUSC",
                "1": "LUAD",
            },
            "source_of_label": "TCGA cohort membership",
        },
        "training_scope": {
            "cohorts": ["TCGA-LUAD", "TCGA-LUSC"],
            "patient_count": int(len(df)),
            "luad_count": int((df["target"] == 1).sum()),
            "lusc_count": int((df["target"] == 0).sum()),
            "external_cohorts_used": False,
        },
        "primary_features": primary_features,
        "race_sensitivity_features": race_features,
        "source_field_mapping": selected_fields,
        "models": final_models,
        "validation_warning": (
            "The TCGA-LUAD cohort contains only LUAD labels and the "
            "TCGA-LUSC cohort contains only LUSC labels. Cohort and target "
            "are perfectly confounded. Random train/test splits and "
            "cross-validation cannot establish independent histology "
            "discrimination. All reported metrics are cohort-confounded "
            "exploratory metrics, not estimates of clinical performance."
        ),
        "excluded_predictors": [
            "ethnicity",
            "KPS",
            "ECOG",
            "ICD-O-3 histology",
            "ICD-O-3 site",
            "ICD-10",
            "histologic_diagnosis",
            "histologic_diagnosis__duplicate_2",
            "treatment",
            "response",
            "outcome",
            "pathological_stage",
            "imaging measurements",
            "Brock malignancy probability",
        ],
        "smoking_status_handling": (
            "Codes 1-5 treated as categorical values; not ordinal. "
            "Unknown, undocumented, discrepancy, and unrecognised codes "
            "are treated as missing."
        ),
        "missing_data_handling": (
            "Numeric predictors median-imputed within each fitted pipeline. "
            "Categorical predictors receive a missing category. "
            "No pack-years are inferred from smoking status."
        ),
        "random_state": RANDOM_STATE,
        "scikit_learn_version": sklearn.__version__,
    }

    joblib.dump(artifact, ARTIFACT_PATH)

    manifest = {
        "model_name": artifact["model_name"],
        "model_version": artifact["model_version"],
        "artifact": str(ARTIFACT_PATH),
        "created_utc": artifact["created_utc"],
        "training_cohorts": ["TCGA-LUAD", "TCGA-LUSC"],
        "expected_patient_counts": EXPECTED,
        "observed_patient_counts": {
            "LUAD": int((df["target"] == 1).sum()),
            "LUSC": int((df["target"] == 0).sum()),
        },
        "primary_predictors": primary_features,
        "algorithms": algorithms,
        "race_analysis": "sensitivity-only",
        "validation": "internal metrics; cohort-confounded",
        "clinical_use": False,
        "api_integration": False,
        "existing_models_modified": False,
        "existing_api_files_modified": False,
        "metrics_file": str(METRICS_PATH),
        "parser_reconciliation_file": str(RECONCILIATION_PATH),
        "missingness_file": str(MISSINGNESS_PATH),
        "warning": artifact["validation_warning"],
    }

    schema = {
        "schema_name": "lunginsight_histology_research_v3",
        "schema_version": "1.0.0",
        "description": (

                "number; source-recorded pack-years; never inferred"
            ),
        },
        "sensitivity_only": {
            "race_sensitivity": "categorical; excluded from primary model"
        },
        "forbidden_predictors": list(artifact["excluded_predictors"]),
        "missing_values": "preserved and handled within model pipelines",
        "clinical_use": False,
        "api_integration": False,
    }

    MANIFEST_PATH.write_text(
        json.dumps(manifest, indent=2),
        encoding="utf-8"
    )
    SCHEMA_PATH.write_text(
        json.dumps(schema, indent=2),
        encoding="utf-8"
    )

    summary = []
    summary.append("LUNGINSIGHT HISTOLOGY RESEARCH MODEL V3")
    summary.append("=" * 60)
    summary.append(f"Created UTC: {artifact['created_utc']}")
    summary.append("")

    summary.append("PARSER EXCLUSIONS")

    )
    summary.append("")
    summary.append("SELECTED SOURCE FIELDS")
    summary.append(json.dumps(selected_fields, indent=2))
    summary.append("")
    summary.append("METRIC INTERPRETATION")
    summary.append(artifact["validation_warning"])
    summary.append("")
    summary.append("ARTIFACT")
    summary.append(str(ARTIFACT_PATH))
    summary.append("")
    summary.append("OUTPUT FILES")
    summary.append(str(METRICS_PATH))
    summary.append(str(RECONCILIATION_PATH))
    summary.append(str(MISSINGNESS_PATH))
    summary.append(str(MANIFEST_PATH))
    summary.append(str(SCHEMA_PATH))
    summary.append("")
    summary.append("API integration: NOT performed.")
    summary.append("Existing v2 and survival artifacts: NOT modified.")

    write_text(SUMMARY_PATH, "\n".join(summary))

    print("\n".join(summary))
    print("\nMODEL METRICS")
    print(metrics_df.to_string(index=False))
